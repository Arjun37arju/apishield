class ScanRequest(BaseModel):
    url: str
    permission_confirmed: bool = False
    check_ids: list[str] | None = None  # None = run all checks
    endpoints: list[str | dict] | None = None
    lab_accounts: bool = False  # use the built-in fake lab test accounts
    auth: dict | None = None  # {"user_a": {...}, "user_b": {...}}, memory only


def _lab_accounts(target):
    """Fake test accounts for the intentionally vulnerable lab only."""
    if not (settings.allow_lab_exception and target["port"] == settings.lab_port):
        raise HTTPException(
            status_code=400,
            detail="lab_accounts only works against the lab on port "
            + str(settings.lab_port)
            + " with ALLOW_LAB_EXCEPTION=true",
        )
    try:
        user_a = BearerTokenProvider("lab-token-user-a")
        user_b = BearerTokenProvider("lab-token-user-b")
        expired = BearerTokenProvider(
            "lab-token-expired", expired_token="lab-token-expired"
        )
    except AuthError as e:
        raise HTTPException(status_code=400, detail=str(e))
    return {"user_a": user_a, "user_b": user_b, "expired_token": expired}


def _provider_from(spec):
    kind = str(spec.get("type", "")).lower()
    if kind == "bearer":
        return BearerTokenProvider(spec.get("token"))
    if kind == "basic":
        return BasicAuthProvider(spec.get("username"), spec.get("password"))
    if kind == "api_key":
        return ApiKeyProvider(
            spec.get("key"), spec.get("header_name") or "X-API-Key"
        )
    raise AuthError("unknown login type")


def _user_accounts(auth):
    """Accounts sent by the dashboard. Held in memory for this scan only."""
    accounts = {}
    try:
        for name in ("user_a", "user_b"):
            spec = (auth or {}).get(name)
            if isinstance(spec, dict) and spec.get("type"):
                accounts[name] = _provider_from(spec)
        a = (auth or {}).get("user_a") or {}
        if str(a.get("type", "")).lower() == "bearer" and a.get("expired_token"):
            accounts["expired_token"] = BearerTokenProvider(
                a.get("expired_token")
            )
    except AuthError as e:
        raise HTTPException(status_code=400, detail=str(e))
    return accounts or None

async def start_scan(req: ScanRequest):
    if not req.permission_confirmed:
        audit("scan_rejected", reason="permission not confirmed")
        raise HTTPException(
            status_code=400,
            detail="You must confirm you have permission to test this target",
        )
    try:
        target = clean_target_url(req.url)
    except TargetError as e:
        audit("scan_rejected", reason=str(e))
        raise HTTPException(status_code=400, detail=str(e))

    if req.endpoints:
        target["endpoints"] = req.endpoints[:50]

    accounts = (
        _lab_accounts(target) if req.lab_accounts else _user_accounts(req.auth)
    )
    check_ids = req.check_ids if req.check_ids else list(registry.keys())
    try:
        scan = manager.create(
            target, req.permission_confirmed, check_ids, accounts=accounts
        )
        manager.start(scan.id)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    return scan.to_dict()

def download_report(file_name: str, download: bool = False):
    # Only a bare .pdf file name inside the reports folder is allowed.
    if Path(file_name).name != file_name or not file_name.endswith(".pdf"):
        raise HTTPException(status_code=400, detail="Invalid file name")
    path = Path(settings.reports_dir) / file_name
    if not path.is_file():
        raise HTTPException(status_code=404, detail="Report not found")
    return FileResponse(path, media_type="application/pdf", filename=file_name, content_disposition_type="attachment" if download else "inline")

import json
import threading
from datetime import datetime, timezone
from pathlib import Path

AUDIT_LOG = Path("logs") / "audit.log"
_lock = threading.Lock()


def audit(event, **details):
    record = {
        "time": datetime.now(timezone.utc).isoformat(),
        "event": event,
        **details,
    }
    with _lock:
        AUDIT_LOG.parent.mkdir(parents=True, exist_ok=True)
        with AUDIT_LOG.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record) + "\n")

SENSITIVE_HEADERS = {
    "authorization", "proxy-authorization", "x-api-key", "cookie", "set-cookie",
}
SENSITIVE_WORDS = ("key", "token", "secret", "password")


def mask_secret(value):
    text = str(value)
    if len(text) <= 8:
        return "****"
    return text[:2] + "****"


def mask_headers(headers):
    out = {}
    for name, value in (headers or {}).items():
        lower = name.lower()
        if lower in SENSITIVE_HEADERS or any(w in lower for w in SENSITIVE_WORDS):
            text = str(value)
            if " " in text:
                scheme, rest = text.split(" ", 1)
                out[name] = scheme + " " + mask_secret(rest)
            else:
                out[name] = mask_secret(text)
        else:
            out[name] = value
    return out

import asyncio
import ipaddress
import socket
import threading
from dataclasses import dataclass, field
from typing import Any
from urllib.parse import unquote, urljoin, urlsplit

import httpx

from app.core.audit import audit
from app.core.settings import settings
from app.core.url_validation import TargetError, clean_target_url

ALLOWED_METHODS = {"GET", "POST", "PUT", "PATCH", "DELETE", "HEAD", "OPTIONS"}
MAX_REDIRECT_HOPS = 3


class RequestBlocked(Exception):
    """The safety gate refused a request. No network call was made."""


@dataclass
class ScanContext:
    scope_host: str
    scope_port: int
    scope_path: str = "/"
    permission_confirmed: bool = False
    max_requests: int = field(default_factory=lambda: settings.max_requests_per_scan)
    requests_made: int = 0
    cancelled: bool = False
    lock: Any = field(default_factory=threading.Lock, repr=False)
    semaphore: Any = field(
        default_factory=lambda: asyncio.Semaphore(settings.max_concurrency),
        repr=False,
    )

    @classmethod
    def from_target(cls, target, permission_confirmed):
        return cls(
            scope_host=target["host"],
            scope_port=target["port"],
            scope_path=target["path"],
            permission_confirmed=permission_confirmed,
        )

    def budget(self):
        return min(self.max_requests, settings.max_requests_per_scan)


@dataclass
class GateResult:
    allowed: bool
    reason: str = ""
    ip: str = ""
    host: str = ""
    port: int = 0
    scheme: str = ""
    path: str = "/"
    query: str = ""


@dataclass
class GateResponse:
    status_code: int
    headers: dict
    body: bytes
    truncated: bool

    @property
    def text(self):
        return self.body.decode("utf-8", errors="replace")


def ip_block_reason(ip_text):
    try:
        ip = ipaddress.ip_address(ip_text.split("%")[0])
    except ValueError:
        return "invalid IP address"
    if ip.version == 6 and ip.ipv4_mapped:
        ip = ip.ipv4_mapped
    if ip.is_loopback:
        return "loopback address"
    if ip.is_link_local:
        return "link-local or cloud metadata address"
    if ip.is_private:
        return "private address"
    if not ip.is_global:
        return "non-public address"
    return None


def default_resolver(host, port):
    infos = socket.getaddrinfo(host, port, type=socket.SOCK_STREAM)
    return sorted({info[4][0] for info in infos})


def _is_ip(text):
    try:
        ipaddress.ip_address(text)
        return True
    except ValueError:
        return False


def _path_in_scope(path, scope_path):
    segments = unquote(path).split("/")
    if ".." in segments:
        return False
    base = scope_path.rstrip("/")
    return base == "" or path == base or path.startswith(base + "/")


def _block(reason, method="", host="", port=0, path=""):
    audit("request_blocked", reason=reason, method=method, host=host,
          port=port, path=path[:200])
    return GateResult(False, reason=reason)


def evaluate(ctx, method, url, resolver=default_resolver):
    method = (method or "").upper()

    # check 1: permission and scope confirmed
    if not ctx.permission_confirmed or not ctx.scope_host:
        return _block("permission or scope not confirmed", method)
    # check 7: scan not cancelled
    if ctx.cancelled:
        return _block("scan was cancelled", method)
    if method not in ALLOWED_METHODS:
        return _block("HTTP method not allowed", method)

    # check 2: URL valid, http/https, allowed port
    try:
        target = clean_target_url(url)
    except TargetError as e:
        return _block(str(e), method)
    host, port, path = target["host"], target["port"], target["path"]
    parts = urlsplit(url.strip())

    # check 4: host, port and path inside the approved scope
    if host.lower() != ctx.scope_host.lower():
        return _block("host is outside the approved scope", method, host, port, path)
    if port != ctx.scope_port:
        return _block("port is outside the approved scope", method, host, port, path)
    if not _path_in_scope(path, ctx.scope_path):
        return _block("path is outside the approved scope", method, host, port, path)

    # check 6: request budget
    if ctx.requests_made >= ctx.budget():
        return _block("request budget used up", method, host, port, path)

    # check 3: DNS result must be an allowed address
    try:
        ips = [host] if _is_ip(host) else resolver(host, port)
    except OSError:
        return _block("host name could not be resolved", method, host, port, path)
    if not ips:
        return _block("host name could not be resolved", method, host, port, path)
    for ip in ips:
        why = ip_block_reason(ip)
        lab_ok = (
            why == "loopback address"
            and settings.allow_lab_exception
            and port == settings.lab_port
        )
        if why and not lab_ok:
            return _block(why + " is not allowed", method, host, port, path)

    with ctx.lock:
        if ctx.requests_made >= ctx.budget():
            return _block("request budget used up", method, host, port, path)
        ctx.requests_made += 1

    return GateResult(True, ip=ips[0], host=host, port=port,
                      scheme=target["scheme"], path=path, query=parts.query)


async def _fetch(ctx, gate, method, headers, json_body, content):
    ip_host = "[" + gate.ip + "]" if ":" in gate.ip else gate.ip
    name = "[" + gate.host + "]" if ":" in gate.host else gate.host
    target = gate.scheme + "://" + ip_host + ":" + str(gate.port) + gate.path
    if gate.query:
        target += "?" + gate.query

    hdrs = {k: v for k, v in (headers or {}).items() if k.lower() != "host"}
    default_port = 443 if gate.scheme == "https" else 80
    hdrs["Host"] = name if gate.port == default_port else name + ":" + str(gate.port)
    extensions = {"sni_hostname": gate.host} if gate.scheme == "https" else {}

    async with ctx.semaphore:
        if ctx.cancelled:
            raise RequestBlocked("scan was cancelled")
        async with asyncio.timeout(settings.request_timeout_seconds):
            async with httpx.AsyncClient(
                timeout=settings.request_timeout_seconds,
                follow_redirects=False,
                trust_env=False,
            ) as client:
                async with client.stream(
                    method, target, headers=hdrs, json=json_body,
                    content=content, extensions=extensions,
                ) as resp:
                    body = bytearray()
                    truncated = False
                    async for chunk in resp.aiter_bytes():
                        body.extend(chunk)
                        if len(body) > settings.max_response_bytes:
                            truncated = True
                            del body[settings.max_response_bytes:]
                            break
                    return GateResponse(resp.status_code, dict(resp.headers),
                                        bytes(body), truncated)


async def send_request(ctx, method, url, headers=None, json_body=None,
                       content=None, resolver=default_resolver,
                       auth=None, auth_variant="valid"):
    """The ONLY function that sends requests to a target."""
    current = url
    hops = 0
    while True:
        gate = await asyncio.to_thread(evaluate, ctx, method, current, resolver)
        if not gate.allowed:
            raise RequestBlocked(gate.reason)
        send_headers = auth.apply(headers, current, auth_variant) if auth else headers
        resp = await _fetch(ctx, gate, method, send_headers, json_body, content)
        location = resp.headers.get("location")
        if settings.follow_redirects and 300 <= resp.status_code < 400 and location:
            hops += 1
            if hops > MAX_REDIRECT_HOPS:
                raise RequestBlocked("too many redirects")
            current = urljoin(current, location)  # check 5: re-checked by the gate
            if resp.status_code in (301, 302, 303):

from urllib.parse import urlsplit

from app.core.settings import settings


class TargetError(ValueError):
    """Raised when a target URL is not acceptable."""


DEFAULT_PORTS = {"http": 80, "https": 443}


def allowed_ports() -> set:
    return {80, 443, settings.lab_port}


def clean_target_url(raw: str) -> dict:
    raw = (raw or "").strip()
    if not raw or len(raw) > 2048:
        raise TargetError("URL is empty or too long")
    if any(ord(c) <= 32 or ord(c) == 127 for c in raw):
        raise TargetError("URL contains spaces or control characters")

    parts = urlsplit(raw)
    scheme = parts.scheme.lower()
    if scheme not in DEFAULT_PORTS:
        raise TargetError("Only http and https are allowed")
    if parts.username or parts.password or "@" in parts.netloc:
        raise TargetError("URLs with a username or password inside are not allowed")

    host = parts.hostname
    if not host:
        raise TargetError("URL has no host")
    try:
        port = parts.port or DEFAULT_PORTS[scheme]
    except ValueError:
        raise TargetError("Invalid port")
    if port not in allowed_ports():
        raise TargetError("Port " + str(port) + " is not allowed")

    shown_host = "[" + host + "]" if ":" in host else host
    path = parts.path or "/"
    return {
        "url": scheme + "://" + shown_host + ":" + str(port) + path,
        "scheme": scheme,
        "host": host,
        "port": port,
        "path": path,
        "scope": shown_host + ":" + str(port),
    }

    p = _db_path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(p))

    try:

    except Exception:

def save_scan(scan, path=None):

             data.get("started_at", ""), data.get("finished_at", ""),

                "INSERT INTO check_results (scan_id, check_id, name, outcome, "
                "message, duration_ms, finding_count) VALUES (?, ?, ?, ?, ?, ?, ?)",

                 _clean(f["endpoint"], 300), f["severity"], f["confidence"],
                 f["status"], _clean(f.get("impact"), 1000), _clean(f.get("fix"), 1000)))

            "SELECT check_id, name, outcome, message, duration_ms, finding_count "

def _text(resp):

class _Acc:
    """Collects results of one check. Merges findings with the same root cause."""

    def __init__(self):
        self.new, self.merged, self.manual, self.tested = [], [], [], 0

    def hit(self, scan, root, title, template, method, url, resp,
            severity, confidence, impact, fix):
        for f in list(scan.findings) + self.new:
            if (f.category == "Authorization" and f.endpoint == template
                    and f.evidence.get("root_cause") == root):
                methods = f.evidence.setdefault("affected_methods", [])
                if method not in methods:
                    methods.append(method)
                self.merged.append(f)
                return
        ev = make_evidence(method, url, {}, resp.status_code, _text(resp))
        ev["root_cause"] = root
        ev["affected_methods"] = [method]
        self.new.append(Finding(
            title=title, category="Authorization", endpoint=template,
            severity=severity, confidence=confidence, impact=impact, fix=fix,
            status=FindingStatus.CONFIRMED if confidence == Confidence.HIGH
            else FindingStatus.SUSPECTED,
            evidence=ev,
        ))

    def report(self, skip_reason):
        if self.new:
            return CheckReport(Outcome.FINDING, "", self.new)
        if self.merged:
            ids = ", ".join(sorted({f.id or "earlier finding" for f in self.merged}))
            return CheckReport(Outcome.PASSED,
                               "weakness found, merged into " + ids + " (same root cause)")
        if self.manual:
            return CheckReport(Outcome.MANUAL, "; ".join(self.manual[:3]))
        if self.tested:
            return CheckReport(Outcome.PASSED)
        return CheckReport(Outcome.SKIPPED, skip_reason)

"""Check family A: Authentication (Step 15).

Five checks, safe GET requests only. Every request goes through
scan.request(), so the safety gate approves it before it is sent.

  1 auth_no_token         endpoint answers with no login at all
  2 auth_invalid_token    a made-up token is accepted
  3 auth_expired_token    an expired token is accepted (needs a test account
                          called "expired_token", otherwise it is Skipped)
  4 auth_malformed_token  a broken token is accepted, or crashes the server
  5 auth_consistency      login is enforced on some endpoints but not others

Endpoints come from scan.target["endpoints"] (a list of {"method", "path"}
dicts or plain path strings). If that key is missing, only the target URL
itself is tested.
"""

# Paths that are normally public on purpose. They are not tested.
PUBLIC_HINTS = ("/login", "/register", "/health", "/docs", "/openapi",
                "/static", "/favicon", "/token")
MAX_ENDPOINTS = 15  # keeps the scan inside the request budget

# Fake tokens. They are not secrets.
INVALID_TOKEN = ("eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9."
                 "eyJzdWIiOiIxIn0.invalidsignature")
MALFORMED_TOKEN = "!!!.@@@.###"

NO_PROTECTED = ("no protected endpoints found: none answered 401 or 403 "
                "when sent with no login")


def is_open(code):
    return 200 <= code < 300


def is_rejected(code):
    return code in (401, 403)


def _path(url):
    return urlsplit(url).path or "/"


def _body(resp):
    try:
        return resp.content
    except Exception:
        return b""


def endpoint_urls(scan):
    """The list of GET endpoint URLs to test (safe, no templates, no public paths)."""
    target_url = scan.target["url"]
    raw = scan.target.get("endpoints")
    if not raw:
        return [target_url]
    parts = urlsplit(target_url)
    base = parts.scheme + "://" + parts.netloc
    urls = []
    for ep in raw:
        if isinstance(ep, str):
            method, path = "GET", ep
        else:
            method = str(ep.get("method", "GET")).upper()
            path = str(ep.get("path", ""))
        if method != "GET" or not path.startswith("/") or "{" in path:
            continue
        if any(hint in path.lower() for hint in PUBLIC_HINTS):
            continue
        url = base + path
        if url not in urls:
            urls.append(url)
    return urls[:MAX_ENDPOINTS]


def _finding(title, url, sent_headers, resp, severity, confidence, status,
             impact, fix):
    return Finding(
        title=title,
        category="Authentication",
        endpoint="GET " + _path(url),
        severity=severity,
        confidence=confidence,
        status=status,
        impact=impact,
        fix=fix,
        evidence=make_evidence("GET", url, sent_headers, resp.status_code,
                               _body(resp)),
    )


class _AuthCheck(BaseCheck):
    category = "Authentication"

    async def baseline(self, scan, url):
        """The answer to a request with no login. Sent once per URL per scan."""
        cache = scan.__dict__.setdefault("_auth_baseline", {})
        if url not in cache:
            cache[url] = await scan.request("GET", url)
        return cache[url]

    async def protected_urls(self, scan):
        """URLs that really ask for a login (they answered 401 or 403)."""
        urls = []
        for url in endpoint_urls(scan):
            resp = await self.baseline(scan, url)
            if is_rejected(resp.status_code):
                urls.append(url)
        return urls

        findings = []
        for url in urls:
            resp = await self.baseline(scan, url)
            if is_open(resp.status_code):
                findings.append(_finding(
                    "Endpoint works with no login", url, {}, resp,

                    "Anyone can call this endpoint without logging in. If it "
                    "is public on purpose this is fine; otherwise its data or "
                    "function is exposed.",
                    "Require a valid login (token or API key) on this endpoint "
                    "and answer 401 when it is missing."))
        if findings:
            return CheckReport(
                Outcome.FINDING,
                str(len(findings)) + " of " + str(len(urls))
                + " endpoints answered with no login",

            Outcome.PASSED,
            "none of the " + str(len(urls)) + " endpoints opened without a login")

class ConsistencyCheck(_AuthCheck):
    id = "auth_consistency"
    name = "Auth 5: login enforced the same on every endpoint"

    async def run(self, scan):

            return CheckReport(Outcome.SKIPPED,
                               "needs at least 2 endpoints to compare")
        open_urls, rejected_urls = [], []
        first_open = None
        for url in urls:
            resp = await self.baseline(scan, url)
            if is_open(resp.status_code):
                open_urls.append(url)
                if first_open is None:
                    first_open = (url, resp)
            elif is_rejected(resp.status_code):
                rejected_urls.append(url)
        if open_urls and rejected_urls:
            shown = ", ".join(_path(u) for u in open_urls[:5])
            finding = _finding(
                "Login is enforced on some endpoints but not others",
                first_open[0], {}, first_open[1],
                Severity.MEDIUM, Confidence.MEDIUM, FindingStatus.MANUAL,
                "Open without login: " + shown + ". Refused without login: "
                + str(len(rejected_urls)) + " other endpoint(s). Check whether "
                "the open ones are public on purpose.",
                "Apply the same login rule to every endpoint, ideally in one "
                "shared place, and list the public endpoints on purpose.")
            return CheckReport(Outcome.FINDING,
                               "login rules differ between endpoints", [finding])
        return CheckReport(
            Outcome.PASSED,
            "login rule was the same on all " + str(len(urls)) + " endpoints")

def _headers(resp):
    try:
        return {str(k).lower(): str(v) for k, v in (resp.headers or {}).items()}
    except Exception:
        return {}


def _text(resp):

def _origin(scan):
    p = urlsplit(scan.target["url"])

def _acct(scan):

def _luhn(digits):
    total = 0
    for i, ch in enumerate(reversed(digits)):
        d = int(ch)
        if i % 2 == 1:
            d *= 2
            if d > 9:
                d -= 9
        total += d
    return total % 10 == 0


def _endpoints(scan):
    """[(path template, url)] for GET endpoints. {id} is filled with a plain 1."""
    raw = scan.target.get("endpoints") or []
    out = []
    for ep in raw:
        if isinstance(ep, str):
            method, path = "GET", ep
        else:
            method = str(ep.get("method", "GET")).upper()
            path = str(ep.get("path", ""))
        if method != "GET" or not path.startswith("/"):
            continue
        if any(h in path.lower() for h in SKIP_HINTS):
            continue
        url = _origin(scan) + ID_PARAM.sub(BASELINE_VALUE, path)
        if (path, url) not in out:
            out.append((path, url))
    if not raw:
        path = urlsplit(scan.target["url"]).path or "/"
        out.append((path, _origin(scan) + path))
    return out[:MAX_ENDPOINTS]

class SecurityHeadersCheck(_ConfigCheck):
    id = "config_headers"
    name = "Config 2: security headers are present"

    async def run(self, scan):
        probe = await _get_probe(scan)
        if not probe:
            return CheckReport(Outcome.SKIPPED, "no endpoint to test")
        https = urlsplit(scan.target["url"]).scheme == "https"
        all_h = [_headers(r) for _, _, r in probe]

        def missing_everywhere(check):
            return all(not check(h) for h in all_h)

        missing = []
        if missing_everywhere(lambda h: "nosniff" in h.get("x-content-type-options", "").lower()):
            missing.append("X-Content-Type-Options: nosniff")
        if missing_everywhere(lambda h: "content-security-policy" in h):
            missing.append("Content-Security-Policy")
        if missing_everywhere(lambda h: "x-frame-options" in h
                              or "frame-ancestors" in h.get("content-security-policy", "").lower()):
            missing.append("X-Frame-Options (or CSP frame-ancestors)")
        if https and missing_everywhere(lambda h: "strict-transport-security" in h):
            missing.append("Strict-Transport-Security")
        if not missing:
            return CheckReport(Outcome.PASSED, "all expected headers were present")
        tpl, url, resp = probe[0]
        sure = Confidence.MEDIUM if len(probe) >= 2 else Confidence.LOW
        finding = _finding(
            "Security headers missing: " + ", ".join(missing), tpl, url, resp,
            Severity.LOW, sure, FindingStatus.SUSPECTED,
            "Missing headers make some browser attacks easier. A missing "
            "header alone is not proof of a weakness (some are not needed for "
            "a pure JSON API): review each one.",
            "Add the missing headers in one shared place for every response.")
        return CheckReport(Outcome.FINDING,
                           str(len(missing)) + " header(s) missing on all "
                           + str(len(probe)) + " endpoint(s) tested", [finding])

class CorsCheck(_ConfigCheck):
    id = "config_cors"
    name = "Config 3: CORS is not too open"

    async def run(self, scan):
        probe = await _get_probe(scan)
        acct = _acct(scan)
        sent = {"Origin": EVIL_ORIGIN}
        hits, tested = [], 0
        for tpl, url, base in probe[:2]:
            if base.status_code == 404:
                continue
            tested += 1
            resp = await scan.request("GET", url, account=acct, headers=sent)
            h = _headers(resp)
            acao = h.get("access-control-allow-origin", "").strip()
            creds = h.get("access-control-allow-credentials", "").strip().lower() == "true"
            if acao == EVIL_ORIGIN:
                kind, rank = "reflects any origin", 3 if creds else 2
            elif acao.lower() == "null":
                kind, rank = "allows the null origin", 2
            elif acao == "*":
                kind, rank = "allows every origin (*)", 1
            else:
                continue
            hits.append((rank, kind, creds, tpl, url, resp))
        if not tested:
            return CheckReport(Outcome.SKIPPED, "no endpoint answered the normal request")
        if not hits:
            return CheckReport(Outcome.PASSED, str(tested)
                               + " endpoint(s) did not allow a foreign origin")
        rank, kind, creds, tpl, url, resp = max(hits, key=lambda x: x[0])
        if rank == 3:
            severity, conf, status = Severity.HIGH, Confidence.HIGH, FindingStatus.CONFIRMED
            impact = ("Any website can call this API with the visitor's login "
                      "and read the answer.")
        else:
            severity, conf, status = Severity.MEDIUM, Confidence.MEDIUM, FindingStatus.SUSPECTED
            impact = ("Websites from any origin can read answers from this "
                      "API. If the API is public on purpose this may be fine; "
                      "otherwise it exposes data.")
        impact += " Seen on " + str(len(hits)) + " of " + str(tested) + " endpoint(s)."
        finding = _finding(
            "CORS " + kind + (" and allows credentials" if creds else ""),
            tpl, url, resp, severity, conf, status, impact,
            "Allow only the exact origins of your own dashboard, never '*' "
            "or a copy of the request origin, and never combine with credentials.",
            sent=sent)
        return CheckReport(Outcome.FINDING, "CORS " + kind, [finding])

def _scrub(text):
    for _, _, rx in SENSITIVE:
        text = rx.sub("[REDACTED]", text)
    return CARD.sub("[REDACTED]", text)


def _sensitive_hits(text):
    hits = [(label, sev) for label, sev, rx in SENSITIVE if rx.search(text)]
    for m in CARD.finditer(text):
        digits = re.sub(r"\D", "", m.group(0))
        if _luhn(digits):
            hits.append(("payment card number", Severity.MEDIUM))
            break
    return hits


class SensitiveDataCheck(_ConfigCheck):
    id = "config_sensitive"
    name = "Config 4: sensitive data in replies"

    async def run(self, scan):
        probe = await _get_probe(scan)
        replies = [p for p in probe if 200 <= p[2].status_code < 300]
        if not replies:
            return CheckReport(Outcome.SKIPPED, "no endpoint answered with a 2xx reply")
        findings = []
        for tpl, url, resp in replies:
            text = _text(resp)
            hits = _sensitive_hits(text)
            if not hits:
                continue
            labels = sorted({label for label, _ in hits})
            worst = max((s for _, s in hits), key=lambda s: SEV_RANK[s])
            only_card = labels == ["payment card number"]
            findings.append(_finding(
                "Reply exposes sensitive data: " + ", ".join(labels), tpl, url, resp,
                worst, Confidence.LOW if only_card else Confidence.MEDIUM,
                FindingStatus.MANUAL if only_card else FindingStatus.SUSPECTED,
                "The reply contains data that should not leave the server. "
                "Values are masked in this report. Check by hand that it is "
                "real data and not an example.",
                "Return only the fields the client needs, never password "
                "hashes, keys or full card numbers.",
                body=_scrub(text)))
        if findings:
            return CheckReport(Outcome.FINDING, str(len(findings))
                               + " endpoint(s) expose sensitive data", findings)
        return CheckReport(Outcome.PASSED, str(len(replies))
                           + " reply(ies) reviewed: nothing sensitive found")

class ErrorDetailCheck(_ConfigCheck):
    id = "config_errors"
    name = "Config 5: errors reveal internal details"

    async def run(self, scan):

        everything = list(probe) + [(NOT_FOUND_PATH, nf_url, nf)]
        errors = [p for p in everything if p[2].status_code >= 400]
        findings, notes = [], []

        found = None
        for tpl, url, resp in errors:
            labels = [label for label, rx in INTERNAL if rx.search(_text(resp))]

                found = (tpl, url, resp, labels)
                break

                      for f in getattr(scan, "findings", []))

                "Turn off debug mode and return a short generic error. Log the "
                "details on the server only."))

                break

        if findings:
            return CheckReport(Outcome.FINDING, "; ".join(notes), findings)
        return CheckReport(Outcome.PASSED, ("; ".join(notes) + ". " if notes else "")

"""Check family C: Input and injection (Step 17).

Five checks, safe GET requests only. Every request goes through
scan.request(), so the safety gate approves it before it is sent.
Run them in this order: C5 reviews what C1 to C4 saw.
"""
import re
from urllib.parse import quote, urlsplit

from app.schemas.findings import (
    Confidence,
    Finding,
    FindingStatus,
    Outcome,
    Severity,
    make_evidence,
)
from app.services.scan_manager import BaseCheck, CheckReport

MAX_ENDPOINTS = 3  # keeps the scan inside the request budget
BASELINE_VALUE = "1"
ID_PARAM = re.compile(r"\{[^}/]+\}")
PUBLIC_HINTS = ("/login", "/register", "/health", "/docs", "/openapi",
                "/static", "/favicon", "/token")

DB_ERROR = re.compile(
    r"(you have an error in your sql syntax|sqlite_error|sqlite3\.\w*error|"
    r"sqlstate\[|psycopg2|pg::\w*error|unterminated quoted string|"
    r"unclosed quotation mark|\bORA-\d{5}\b|mysql_fetch|"
    r"syntax error at or near|System\.Data\.SqlClient|sequelize\w*error|"
    r"sql syntax)", re.IGNORECASE)
TRACE = re.compile(
    r"(Traceback \(most recent call last\)|\bat [\w$.]+\([\w$.]+\.java:\d+\)|"
    r"node_modules[\\/]|System\.\w+Exception|Stack trace:|Werkzeug Debugger|"
    r"File \"[^\"]+\", line \d+)", re.IGNORECASE)


def _signature(text):
    if DB_ERROR.search(text or ""):
        return "database error"
    if TRACE.search(text or ""):
        return "stack trace"
    return ""


def _text(resp):
    try:
        return resp.text or ""
    except Exception:
        return ""


def _origin(scan):
    p = urlsplit(scan.target["url"])
    return p.scheme + "://" + p.netloc


def _acct(scan):
    return "user_a" if "user_a" in scan.accounts else None


def _seen(scan):
    return scan.__dict__.setdefault("_input_seen", [])


def _reported(scan):
    return scan.__dict__.setdefault("_input_reported", set())


def _targets(scan):
    """GET endpoints to test: [(path template, query parameter name)]."""
    raw = scan.target.get("endpoints") or []
    out = []
    for ep in raw:
        if isinstance(ep, str):
            method, path, params = "GET", ep, []
        else:
            method = str(ep.get("method", "GET")).upper()
            path = str(ep.get("path", ""))
            params = ep.get("params") or []
        if method != "GET" or not path.startswith("/"):
            continue
        if any(h in path.lower() for h in PUBLIC_HINTS):
            continue
        param = params[0] if params and isinstance(params[0], str) else "id"
        if (path, param) not in out:
            out.append((path, param))
    if not raw:
        out.append((urlsplit(scan.target["url"]).path or "/", "id"))
    out.sort(key=lambda t: "{" not in t[0])  # path parameters first
    return out[:MAX_ENDPOINTS]


def _url_value(scan, tpl, param, value):
    """Put the test value in the path parameter, or in a query parameter."""
    if ID_PARAM.search(tpl):
        first = [True]

        def sub(_m):
            if first[0]:
                first[0] = False
                return quote(value, safe="")
            return BASELINE_VALUE
        return _origin(scan) + ID_PARAM.sub(sub, tpl)
    return _origin(scan) + tpl + "?" + param + "=" + quote(value, safe="")


def _url_query(scan, tpl, query):
    path = ID_PARAM.sub(BASELINE_VALUE, tpl)
    return _origin(scan) + path + "?" + query


async def _baseline(scan, tpl, param, acct):
    cache = scan.__dict__.setdefault("_input_baseline", {})
    if tpl not in cache:
        url = _url_value(scan, tpl, param, BASELINE_VALUE)
        cache[tpl] = await scan.request("GET", url, account=acct)
    return cache[tpl]


def _finding(title, tpl, url, resp, severity, confidence, status, impact, fix):
    return Finding(
        title=title,
        category="Input and injection",
        endpoint="GET " + tpl,
        severity=severity,
        confidence=confidence,
        status=status,
        impact=impact,
        fix=fix,
        evidence=make_evidence("GET", url, {}, resp.status_code, _text(resp)),
    )


class _InputCheck(BaseCheck):
    category = "Input and injection"
    values = ()          # [(label, value)]
    raw_query = False    # True: values are whole query strings
    title = ""
    impact = ""
    fix = ""
    sig_severity = Severity.MEDIUM

    async def run(self, scan):
        targets = _targets(scan)
        if not targets:
            return CheckReport(Outcome.SKIPPED, "no GET endpoints to test")
        acct = _acct(scan)
        findings, tested = [], 0
        for tpl, param in targets:
            base = await _baseline(scan, tpl, param, acct)
            if base.status_code in (401, 403, 404):
                continue
            tested += 1
            base_sig = _signature(_text(base))
            hit, weak = False, None
            for label, value in self.values:
                if self.raw_query:
                    url = _url_query(scan, tpl, value)
                else:
                    url = _url_value(scan, tpl, param, value)
                resp = await scan.request("GET", url, account=acct)
                _seen(scan).append((tpl, url, resp))
                sig = _signature(_text(resp))
                if sig and sig != base_sig:
                    findings.append(_finding(
                        self.title + " (" + sig + ", input: " + label + ")",
                        tpl, url, resp, self.sig_severity, Confidence.MEDIUM,
                        FindingStatus.SUSPECTED, self.impact, self.fix))
                    _reported(scan).add(tpl)
                    hit = True
                    break
                if resp.status_code >= 500 and base.status_code < 500 and weak is None:
                    weak = (url, resp, label)
            if not hit and weak:
                url, resp, label = weak
                findings.append(_finding(
                    "Server error on unusual input (" + label + ")",
                    tpl, url, resp, Severity.LOW, Confidence.LOW,
                    FindingStatus.MANUAL,
                    "The server failed instead of refusing the input. An odd "
                    "response alone is not proof of a weakness: review it by hand.",
                    self.fix))
        if findings:
            return CheckReport(Outcome.FINDING,
                               str(len(findings)) + " endpoint(s) need attention",
                               findings)
        if tested:
            return CheckReport(Outcome.PASSED,
                               str(tested) + " endpoint(s) handled the input safely")
        return CheckReport(
            Outcome.SKIPPED,
            "no endpoint answered the normal request (it may need a login: "
            "give the scan a user_a account)")

"""Check family D: Rate limiting (Step 18).

Four checks, safe GET requests only, one at a time. Every request goes
through scan.request(), so the safety gate approves it before it is sent.
The endpoints are probed once per scan and the four checks share the result.
"""
import asyncio
from urllib.parse import urlsplit

from app.core.safety_gate import RequestBlocked
from app.schemas.findings import (
    Confidence,
    Finding,
    FindingStatus,
    Outcome,
    Severity,
    make_evidence,
)
from app.services.scan_manager import BaseCheck, CheckReport

BURST = 20          # most requests sent to one endpoint
MAX_ENDPOINTS = 2   # keeps the scan far inside the 100-request budget
MAX_WAIT = 10       # longest pause before the recovery test, in seconds
DEFAULT_WAIT = 5
SKIP_HINTS = ("/docs", "/openapi", "/static", "/favicon", "/health")
LOGIN_HINTS = ("login", "token", "auth", "signin", "register", "reset", "password")
LIMIT_HEADERS = ("x-ratelimit-limit", "ratelimit-limit", "x-rate-limit-limit",
                 "x-ratelimit-remaining", "ratelimit-remaining")
NO_ENDPOINTS = "no endpoint could be tested (needs a GET endpoint without an {id})"


async def _sleep(seconds):
    await asyncio.sleep(seconds)


def _headers(resp):
    try:
        return {str(k).lower(): str(v) for k, v in (resp.headers or {}).items()}
    except Exception:
        return {}


def _text(resp):
    try:
        return resp.text or ""
    except Exception:
        return ""


def _origin(scan):
    p = urlsplit(scan.target["url"])
    return p.scheme + "://" + p.netloc


def _acct(scan):
    return "user_a" if "user_a" in scan.accounts else None


def _cancelled(scan):
    return bool(getattr(getattr(scan, "ctx", None), "cancelled", False))

_POST = {}  # path -> JSON body, filled in by _targets for POST login endpoints
LOGIN_TEST_BODY = {"username": "api-gapwolf-test", "password": "wrong-password-for-test"}

def _is_login_like(path):
    return any(h in path.lower() for h in LOGIN_HINTS)


def _targets(scan):
    """Paths to test (no {id} templates). GET endpoints, plus login-like POST
    endpoints. Login-like paths first."""
    raw = scan.target.get("endpoints") or []
    found = []
    _POST.clear()
    for ep in raw:
        if isinstance(ep, str):
            method, path, body = "GET", ep, None
        else:
            method = str(ep.get("method", "GET")).upper()
            path = str(ep.get("path", ""))
            body = ep.get("json")
        if not path.startswith("/") or "{" in path:
            continue
        if any(h in path.lower() for h in SKIP_HINTS):
            continue
        if method == "POST":
            if not _is_login_like(path):
                continue  # only login-like POST endpoints are rate tested
            _POST[path] = body or LOGIN_TEST_BODY
        elif method != "GET":
            continue
        if path not in found:
            found.append(path)
    if not raw:
        found.append(urlsplit(scan.target["url"]).path or "/")
    found.sort(key=lambda p: not _is_login_like(p))
    return found[:MAX_ENDPOINTS]


async def _probe_endpoint(scan, path, acct):
    url = _origin(scan) + path
    method = "POST" if path in _POST else "GET"
    info = {"path": path, "url": url, "sent": 0, "first": None, "last": None,
            "limited_at": None, "retry_after": None, "other_at": None,
            "header_limit": None, "has_headers": False, "missing": False,
            "stopped": ""}
    for i in range(1, BURST + 1):
        if _cancelled(scan):
            info["stopped"] = "scan cancelled"
            break
        try:
            resp = await scan.request(
                method, url,
                account=None if method == "POST" else acct,
                json_body=_POST.get(path),
            )
        except RequestBlocked as e:
            if info["sent"] == 0:
                raise
            info["stopped"] = "safety gate: " + str(e)
            break
        info["sent"] = i
        info["last"] = resp
        code = resp.status_code
        hdr = _headers(resp)
        if info["first"] is None:
            info["first"] = code
        if i == 1 and code == 404:
            info["missing"] = True
            break
        for name in LIMIT_HEADERS:
            if name in hdr:
                info["has_headers"] = True
                if name.endswith("limit") and info["header_limit"] is None:
                    info["header_limit"] = hdr[name]
        if code == 429:
            info["limited_at"] = i
            info["retry_after"] = hdr.get("retry-after")
            break  # stop as soon as the server says "too many requests"
        first = info["first"]
        if code in (403, 503) and i > 1 and first is not None and 200 <= first < 300:
            info["other_at"] = i
            break
    return info


async def _get_probe(scan):
    probes = scan.__dict__.get("_rate_probe")
    if probes is None:
        acct = _acct(scan)
        probes = []
        for path in _targets(scan):
            info = await _probe_endpoint(scan, path, acct)
            if info["sent"] and not info["missing"]:
                probes.append(info)
        scan.__dict__["_rate_probe"] = probes
    return probes


def _reacted(p):
    return bool(p["limited_at"] or p["other_at"])


def _wait_seconds(retry_after):
    try:
        return max(1, min(MAX_WAIT, int(float(retry_after))))
    except (TypeError, ValueError):
        return DEFAULT_WAIT


def _finding(title, path, url, resp, severity, confidence, status, impact, fix):
    method = "POST" if path in _POST else "GET"
    return Finding(
        title=title,
        category="Rate limiting",
        endpoint=method + " " + path,
        severity=severity,
        confidence=confidence,
        status=status,
        impact=impact,
        fix=fix,
        evidence=make_evidence(method, url, {}, resp.status_code, _text(resp))
    )


class _RateCheck(BaseCheck):
    category = "Rate limiting"


class RateLimitPresentCheck(_RateCheck):
    id = "rate_present"
    name = "Rate 1: endpoint has a request limit"

    async def run(self, scan):
        probes = await _get_probe(scan)
        if not probes:
            return CheckReport(Outcome.SKIPPED, NO_ENDPOINTS)
        findings = []
        for p in probes:
            if _reacted(p) or p["has_headers"]:
                continue
            findings.append(_finding(
                "No request limit seen on " + p["path"] + " (after "
                + str(p["sent"]) + " requests)",
                p["path"], p["url"], p["last"],
                Severity.MEDIUM if _is_login_like(p["path"]) else Severity.LOW,
                Confidence.LOW, FindingStatus.SUSPECTED,
                str(p["sent"]) + " requests in a row were all accepted: no 429 "
                "and no rate-limit headers. Without a limit, attackers can "
                "guess passwords or overload the service. A limit above this "
                "safe test size may exist: confirm by hand.",
                "Add a rate limit per user and per IP address (for example a "
                "few login tries per minute) and answer 429 Too Many Requests "
                "with a Retry-After header."))
        if findings:
            return CheckReport(
                Outcome.FINDING,
                str(len(findings)) + " of " + str(len(probes))
                + " endpoint(s) showed no limit", findings)
        return CheckReport(Outcome.PASSED,
                           "all " + str(len(probes)) + " endpoint(s) show a limit")


class RateThresholdCheck(_RateCheck):
    id = "rate_threshold"
    name = "Rate 2: when the limit reacts"

    async def run(self, scan):
        probes = await _get_probe(scan)
        msgs = []
        for p in probes:
            if p["limited_at"]:
                msgs.append(p["path"] + ": 429 after " + str(p["limited_at"]) + " requests")
            elif p["other_at"]:
                msgs.append(p["path"] + ": status " + str(p["last"].status_code)
                            + " after " + str(p["other_at"]) + " requests")
            elif p["header_limit"]:
                msgs.append(p["path"] + ": declares a limit of " + p["header_limit"])
        if msgs:
            return CheckReport(Outcome.PASSED, "; ".join(msgs))
        return CheckReport(
            Outcome.SKIPPED,
            "no limit reacted within " + str(BURST) + " safe requests, so there "
            "is no threshold to measure")


class Http429Check(_RateCheck):
    id = "rate_http_429"
    name = "Rate 3: limit answers with HTTP 429"

    async def run(self, scan):
        probes = await _get_probe(scan)
        reacted = [p for p in probes if _reacted(p)]
        if not reacted:
            return CheckReport(Outcome.SKIPPED,
                               "no limit reacted, so there is no answer to check")
        findings, notes = [], []
        for p in reacted:
            if p["limited_at"]:
                notes.append(p["path"] + ": 429, Retry-After "
                             + ("present" if p["retry_after"] else "missing"))
                continue
            findings.append(_finding(
                "Limit answers with status " + str(p["last"].status_code)
                + " instead of 429",
                p["path"], p["url"], p["last"], Severity.LOW, Confidence.LOW,
                FindingStatus.SUSPECTED,
                "The server stops answering after many requests but does not "
                "use the standard 429 status, so clients and tools cannot "
                "tell a limit from a real error.",

class RecoveryCheck(_RateCheck):
    id = "rate_recovery"
    name = "Rate 4: endpoint works again after the limit"

    async def run(self, scan):
        probes = await _get_probe(scan)
        limited = [p for p in probes if p["limited_at"]]
        if not limited:
            return CheckReport(Outcome.SKIPPED,
                               "no 429 was seen, so there is nothing to recover from")
        acct = _acct(scan)
        findings, msgs = [], []
        for p in limited:
            wait = _wait_seconds(p["retry_after"])
            resp, waited = None, 0
            for _ in range(2):  # at most 2 extra requests per endpoint
                if _cancelled(scan):
                    break
                await _sleep(wait)
                waited += wait
                resp = await scan.request("GET", p["url"], account=acct)
                if resp.status_code != 429:
                    break
            if resp is None:
                continue
            if resp.status_code != 429:
                msgs.append(p["path"] + ": works again after about "
                            + str(waited) + " s")
                continue
            findings.append(_finding(
                "Limit still active after " + str(waited) + " seconds",
                p["path"], p["url"], resp, Severity.LOW, Confidence.LOW,
                FindingStatus.MANUAL,
                "The endpoint was still blocked after waiting. This may be a "
                "long lockout (fine) or a limit that never resets (a problem "
                "for real users). Review by hand.",
                "Reset the limit after a short window and tell clients how "
                "long to wait with a Retry-After header."))
        if findings:
            return CheckReport(Outcome.FINDING,
                               "an endpoint stayed blocked", findings)
        if msgs:
            return CheckReport(Outcome.PASSED, "; ".join(msgs))
        return CheckReport(Outcome.SKIPPED, "scan cancelled before the recovery test")

HEADER_NAME = re.compile(r"[A-Za-z0-9-]{1,64}")
VARIANTS = ("valid", "none", "wrong")


class AuthError(ValueError):

def _clean(value, label):
    text = str(value or "")
    if not text or any(ord(c) < 32 or ord(c) == 127 for c in text):
        raise AuthError(label + " is empty or has control characters")
    return text


def _lab_http_allowed(url):
    parts = urlsplit(url)
    return (
        settings.allow_lab_exception
        and parts.port == settings.lab_port
        and parts.hostname in ("127.0.0.1", "localhost", "::1")
    )


class AuthProvider:
    """One interface for every login method. Checks never contain login code."""

    name = "base"

    def apply(self, headers, url, variant="valid"):
        raise NotImplementedError

    def __repr__(self):
        return "<" + self.__class__.__name__ + " credentials hidden>"


class BasicAuthProvider(AuthProvider):
    name = "basic"

    def __init__(self, username, password):
        self._username = _clean(username, "username")
        if ":" in self._username:
            raise AuthError("username must not contain a colon")
        self._password = _clean(password, "password")

    def apply(self, headers, url, variant="valid"):
        if variant not in VARIANTS:
            raise AuthError("unknown variant")
        out = dict(headers or {})
        if variant == "none":
            return out
        if urlsplit(url).scheme != "https" and not _lab_http_allowed(url):
            raise AuthError("Basic Auth is only allowed over HTTPS")
        password = self._password if variant == "valid" else "invalid-password"
        raw = (self._username + ":" + password).encode("utf-8")
        out["Authorization"] = "Basic " + base64.b64encode(raw).decode("ascii")
        return out


class ApiKeyProvider(AuthProvider):
    name = "api_key"

    def __init__(self, key, header_name="X-API-Key"):
        if not HEADER_NAME.fullmatch(str(header_name or "")):
            raise AuthError("invalid header name")
        self._header = header_name
        self._key = _clean(key, "API key")

    def apply(self, headers, url, variant="valid"):
        if variant not in VARIANTS:
            raise AuthError("unknown variant")
        out = dict(headers or {})
        if variant == "none":
            return out
        out[self._header] = self._key if variant == "valid" else "invalid-key"
        return out

BEARER_VARIANTS = ("valid", "none", "wrong", "expired", "malformed")
SKEW_SECONDS = 10


def _tamper(token):
    """Make a well-formed but invalid token (bad signature for a JWT)."""
    if token.count(".") == 2:
        head, sig = token.rsplit(".", 1)
        if sig:
            last = "A" if sig[-1] != "A" else "B"
            return head + "." + sig[:-1] + last
    return "invalid-token"


class _BearerBase(AuthProvider):
    def __init__(self, expired_token=None):
        self._expired = _clean(expired_token, "expired token") if expired_token else None

    def _current_token(self):
        raise NotImplementedError

    def apply(self, headers, url, variant="valid"):
        if variant not in BEARER_VARIANTS:
            raise AuthError("unknown variant")
        out = dict(headers or {})
        if variant == "none":
            return out
        if urlsplit(url).scheme != "https" and not _lab_http_allowed(url):
            raise AuthError("Tokens are only sent over HTTPS")
        if variant == "expired":
            if not self._expired:
                raise AuthError("no expired token was supplied")
            token = self._expired
        elif variant == "malformed":
            token = "!!malformed!!"
        else:
            real = self._current_token()
            token = real if variant == "valid" else _tamper(real)
        out["Authorization"] = "Bearer " + token
        return out

class OAuthClientCredentialsProvider(_BearerBase):
    """OAuth 2.0 client credentials flow only. Use a least-privilege test client."""

    name = "oauth2_client_credentials"

    def __init__(self, token_url, client_id, client_secret, scope="",
                 expired_token=None):
        super().__init__(expired_token)
        self._token_url = _clean(token_url, "token URL")
        self._client_id = _clean(client_id, "client id")
        self._client_secret = _clean(client_secret, "client secret")
        self._scope = str(scope or "").strip()
        if any(ord(c) < 32 or ord(c) == 127 for c in self._scope):
            raise AuthError("scope has control characters")
        self._access_token = None
        self._expires_at = 0.0
        self.granted_scope = ""

    def _current_token(self):
        if not self._access_token:
            raise AuthError("no access token yet: call prepare() first")
        return self._access_token

    def needs_token(self):
        return (not self._access_token
                or time.monotonic() >= self._expires_at - SKEW_SECONDS)

    async def prepare(self, ctx, resolver=default_resolver):
        """Get (or renew) the access token. The request goes through the gate."""
        if not self.needs_token():
            return
        if urlsplit(self._token_url).scheme != "https" and not _lab_http_allowed(self._token_url):
            raise AuthError("The token endpoint must use HTTPS")
        form = {
            "grant_type": "client_credentials",
            "client_id": self._client_id,
            "client_secret": self._client_secret,
        }
        if self._scope:
            form["scope"] = self._scope
        resp = await send_request(
            ctx, "POST", self._token_url,
            headers={"Content-Type": "application/x-www-form-urlencoded",
                     "Accept": "application/json"},
            content=urlencode(form),
            resolver=resolver,
        )
        if resp.status_code != 200:
            raise AuthError("token request failed with status " + str(resp.status_code))
        if resp.truncated:
            raise AuthError("token response is too large")
        try:
            data = json.loads(resp.body.decode("utf-8"))
        except (ValueError, UnicodeDecodeError):
            raise AuthError("token response is not valid JSON")
        if not isinstance(data, dict) or not isinstance(data.get("access_token"), str):
            raise AuthError("token response has no access_token")
        token = _clean(data["access_token"], "access token")
        if str(data.get("token_type", "")).lower() != "bearer":
            raise AuthError("unsupported token type")
        try:
            expires_in = int(data.get("expires_in", 300))
        except (TypeError, ValueError):
            expires_in = 300
        expires_in = max(1, min(expires_in, 86400))
        self._access_token = token
        self._expires_at = time.monotonic() + expires_in
        self.granted_scope = str(data.get("scope", ""))[:200]

import json

from app.core.settings import settings


class OpenAPIError(ValueError):
    """Raised when an uploaded API description is not acceptable."""


ALLOWED_EXTENSIONS = (".json", ".yaml", ".yml")
HTTP_METHODS = ("get", "post", "put", "patch", "delete", "head", "options")
MAX_ENDPOINTS = 1000


def _check_and_decode(filename, content):
    name = (filename or "").lower()
    if not name.endswith(ALLOWED_EXTENSIONS):
        raise OpenAPIError("Only .json, .yaml and .yml files are allowed")
    if len(content) > settings.max_upload_bytes:
        raise OpenAPIError("File is too large")
    try:
        return name, content.decode("utf-8")
    except UnicodeDecodeError:
        raise OpenAPIError("File is not valid UTF-8 text")


def _parse_json(text):
    try:
        return json.loads(text)
    except (ValueError, RecursionError):
        raise OpenAPIError("File is not valid JSON")


def _parse_yaml(text):
    try:
        for token in yaml.scan(text, Loader=yaml.SafeLoader):
            if isinstance(token, yaml.AliasToken):
                raise OpenAPIError("YAML aliases are not allowed")
        return yaml.safe_load(text)
    except OpenAPIError:
        raise
    except (yaml.YAMLError, RecursionError):
        raise OpenAPIError("File is not valid or safe YAML")


def parse_openapi(filename, content):
    name, text = _check_and_decode(filename, content)
    doc = _parse_json(text) if name.endswith(".json") else _parse_yaml(text)

    if not isinstance(doc, dict) or not ("openapi" in doc or "swagger" in doc):
        raise OpenAPIError("File is not an OpenAPI / Swagger document")
    paths = doc.get("paths")
    if not isinstance(paths, dict) or not paths:
        raise OpenAPIError("No paths found in the document")

    endpoints = []
    for path, item in paths.items():
        if not isinstance(path, str) or not path.startswith("/"):
            raise OpenAPIError("A path in the document is invalid")
        if not isinstance(item, dict):
            continue
        for method in HTTP_METHODS:
            op = item.get(method)
            if isinstance(op, dict):
                endpoints.append({
                    "method": method.upper(),
                    "path": path[:300],
                    "summary": str(op.get("summary", ""))[:200],
                })
                if len(endpoints) > MAX_ENDPOINTS:
                    raise OpenAPIError("Too many endpoints")
    if not endpoints:
        raise OpenAPIError("No endpoints found in the document")

    info = doc.get("info") if isinstance(doc.get("info"), dict) else {}
    return {
        "title": str(info.get("title", ""))[:200],
        "version": str(info.get("version", ""))[:50],
        "endpoint_count": len(endpoints),
        "endpoints": endpoints,
    }

class ScanManager:
    def __init__(self, registry, on_finished=None):
        self.registry = registry  # {check_id: check object}
        self.on_finished = on_finished  # called once when a scan ends
        self.scans = {}

    def create(self, target, permission_confirmed, check_ids, accounts=None,
               resolver=default_resolver):
        if not permission_confirmed:
            raise ValueError("permission not confirmed")
        unknown = [c for c in check_ids if c not in self.registry]
        if unknown:
            raise ValueError("unknown check: " + ", ".join(unknown))
        ctx = ScanContext.from_target(target, permission_confirmed)
        scan = Scan(target=target, ctx=ctx, check_ids=list(check_ids),
                    accounts=dict(accounts or {}), resolver=resolver)
        self.scans[scan.id] = scan
        audit("scan_created", scan_id=scan.id, scope=target["scope"],
              checks=len(scan.check_ids))
        return scan

    def get(self, scan_id):
        scan = self.scans.get(scan_id)
        if scan is None:
            raise KeyError("scan not found")
        return scan

    def start(self, scan_id):
        scan = self.get(scan_id)
        if scan.state != ScanState.QUEUED:
            raise ValueError("scan cannot be started from state " + scan.state.value)
        scan.state = ScanState.RUNNING
        scan.task = asyncio.create_task(self._run(scan))
        return scan

    async def wait(self, scan_id):
        scan = self.get(scan_id)
        if scan.task is not None:
            await asyncio.wait({scan.task})
        return scan

    def cancel(self, scan_id):
        scan = self.get(scan_id)
        if scan.state in FINAL_STATES:
            return scan
        scan.ctx.cancelled = True  # the gate now blocks every request
        scan.state = ScanState.CANCELLED
        scan.finished_at = _now()
        if scan.task is not None and not scan.task.done():
            scan.task.cancel()
        audit("scan_cancelled", scan_id=scan.id, requests=scan.ctx.requests_made)
        return scan

    async def _run(self, scan):
        scan.state = ScanState.RUNNING
        scan.started_at = _now()
        audit("scan_started", scan_id=scan.id)
        try:
            for check_id in scan.check_ids:
                if scan.ctx.cancelled:
                    break
                await self._run_check(scan, self.registry[check_id])
            if scan.state == ScanState.RUNNING:
                scan.state = ScanState.COMPLETED
        except asyncio.CancelledError:
            scan.state = ScanState.CANCELLED
            raise
        except Exception as e:
            scan.state = ScanState.FAILED
            scan.error = type(e).__name__
        finally:
            scan.finished_at = scan.finished_at or _now()
            audit("scan_finished", scan_id=scan.id, state=scan.state.value,
                  requests=scan.ctx.requests_made)
            if self.on_finished is not None:
                try:
                    self.on_finished(scan)
                except Exception as e:
                    audit("scan_save_failed", scan_id=scan.id, error=type(e).__name__)

    async def _run_check(self, scan, check):
        started = time.monotonic()
        findings = []
        missing = [a for a in check.requires if a not in scan.accounts]
        if missing:
            outcome = Outcome.SKIPPED
            message = "test account missing: " + ", ".join(missing)
        else:
            try:
                report = await check.run(scan)
                outcome, message, findings = report.outcome, report.message, report.findings
            except RequestBlocked as e:
                outcome, message = Outcome.ERROR, "blocked by the safety gate: " + str(e)
            except AuthError as e:
                outcome, message = Outcome.ERROR, "login problem: " + str(e)
            except (httpx.HTTPError, TimeoutError) as e:
                outcome, message = Outcome.ERROR, "request failed: " + type(e).__name__
            except Exception as e:
                outcome, message = Outcome.ERROR, "internal error: " + type(e).__name__

        if findings and outcome != Outcome.FINDING:
            outcome = Outcome.FINDING
        if outcome == Outcome.FINDING and not findings:
            outcome, message = Outcome.ERROR, "check reported a finding without details"
        for finding in findings:
            scan.add_finding(finding)
        scan.results.append(CheckResult(
            check_id=check.id, name=check.name, outcome=outcome, message=message,
            duration_ms=int((time.monotonic() - started) * 1000),
            finding_count=len(findings),
        ))

        self.calls.append((method, url, account))
        return self.handler(method, url, account)

    def add_finding(self, f):
        f.id = "API-%03d" % (len(self.findings) + 1)
        self.findings.append(f)


def run_all(scan):
    async def go():
        out = []
        for check in AUTHZ_CHECKS:
            missing = [a for a in check.requires if a not in scan.accounts]
            if missing:
                out.append(Outcome.SKIPPED)
                continue
            rep = await check.run(scan)
            for f in rep.findings:
                scan.add_finding(f)
            out.append(rep.outcome)
        return out
    return asyncio.run(go())

def limited(limit_after=5, status=429, recovers=True):
    counts = {}

    def handler(method, url):
        n = counts[url] = counts.get(url, 0) + 1
        if n > limit_after and not (recovers and SLEPT):
            return R(status, "Too many requests", {"Retry-After": "2"})
        return R(200, "ok")
    return handler

async function request(path, options = {}) {
  const res = await fetch(BASE_URL + path, {
    headers: { "Content-Type": "application/json" },
    ...options,
  });
  let data = null;
  try {
    data = await res.json();
  } catch {
    data = null;
  }
  if (!res.ok) {
    const detail = data && data.detail;
    throw new Error(
      typeof detail === "string" ? detail : "Request failed (" + res.status + ")"
    );
  }
  return data;
}

function specFrom(a) {
  if (!a) return null;
  if (a.type === "bearer" && a.token) {
    return { type: "bearer", token: a.token, expired_token: a.expired_token || "" };
  }
  if (a.type === "basic" && a.username && a.password) {
    return { type: "basic", username: a.username, password: a.password };
  }
  if (a.type === "api_key" && a.key) {
    return { type: "api_key", key: a.key, header_name: a.header_name || "X-API-Key" };
  }
  return null;
}

export function buildAuthPayload() {
  const user_a = specFrom(authForm.user_a);
  if (!user_a) return null;
  const out = { user_a };
  const user_b = specFrom(authForm.user_b);
  if (user_b) out.user_b = user_b;
  return out;
}

import { useEffect, useState } from "react";
import { listChecks, getSelectedChecks, saveSelectedChecks } from "../api/client";

export default function ScanProfile() {
  const [checks, setChecks] = useState([]);
  const [selected, setSelected] = useState([]);
  const [error, setError] = useState("");

  useEffect(() => {
    listChecks()
      .then((list) => {
        setChecks(list);
        const stored = getSelectedChecks();
        setSelected(stored || list.map((c) => c.id));
      })
      .catch((e) => setError(e.message));
  }, []);

  function update(ids) {
    setSelected(ids);
    // everything selected = store nothing, so new checks are included later
    saveSelectedChecks(ids.length === checks.length ? null : ids);
  }

  function toggle(id) {
    update(selected.includes(id) ? selected.filter((x) => x !== id) : [...selected, id]);
  }

  return (
    <div style={{ maxWidth: 760 }}>
      <h2>Scan Profile</h2>
      <p>Choose which checks the next scan runs. Your choice is used by Target Setup.</p>
      {error && <p style={{ color: "red" }}>{error}</p>}

      <p>
        <button onClick={() => update(checks.map((c) => c.id))}>Select all</button>{" "}
        <button onClick={() => update([])}>Clear</button>{" "}
        {selected.length} of {checks.length} selected
      </p>
      {selected.length === 0 && checks.length > 0 && (
        <p style={{ color: "orange" }}>
          Nothing selected: the scan will run ALL checks.
        </p>
      )}

      <table border="1" cellPadding="4" style={{ borderCollapse: "collapse", width: "100%" }}>
        <thead>
          <tr><th>Run</th><th>Check</th><th>What it does</th></tr>
        </thead>

import { useEffect, useState } from "react";

const DONE = ["Completed", "Cancelled", "Failed"];

export default function ScanProgress() {
  const params = useParams();
  const id = params.id || lastScan();
  const [scan, setScan] = useState(null);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!id) return undefined;
    let stop = false;
    let timer = null;
    async function poll() {
      try {
        const data = await getScan(id);
        if (stop) return;
        setScan(data);
        setError("");
        if (!DONE.includes(data.state)) timer = setTimeout(poll, 1000);
      } catch (err) {
        if (!stop) setError(err.message);
      }
    }
    poll();
    return () => {
      stop = true;
      if (timer) clearTimeout(timer);
    };
  }, [id]);

  async function handleCancel() {
    try {
      setScan(await cancelScan(id));
    } catch (err) {
      setError(err.message);
    }
  }

  if (!id) {
    return (
      <div>
        <h2>Scan Progress</h2>
        <p>No scan yet. Start one on the <Link to="/">Target Setup</Link> screen.</p>
      </div>
    );
  }

  const running = scan && !DONE.includes(scan.state);

  return (
    <div>
      <h2>Scan Progress</h2>
      {error && <p style={{ color: "red" }}>{error}</p>}
      {scan && (

import { useState } from "react";
import { useNavigate } from "react-router-dom";
import {
  validateTarget,
  startScan,
  rememberScan,
  getSelectedChecks,
  buildAuthPayload,
} from "../api/client";

const METHODS = ["GET", "POST", "PUT", "PATCH", "DELETE"];

// One endpoint per line:  /api/x   or   METHOD /path   or   GET /api/search q
function parseEndpoints(text) {
  const out = [];
  for (const raw of text.split("\n")) {
    const parts = raw.trim().split(/\s+/).filter(Boolean);
    if (!parts.length) continue;
    if (METHODS.includes(parts[0].toUpperCase()) && parts[1]) {
      const item = { method: parts[0].toUpperCase(), path: parts[1] };
      if (parts.length > 2) item.params = parts.slice(2);
      out.push(item);
    } else {
      out.push(parts[0]);
    }
  }
  return out;
}

export default function TargetSetup() {
  const navigate = useNavigate();
  const [url, setUrl] = useState("");
  const [permission, setPermission] = useState(false);
  const [endpointsText, setEndpointsText] = useState("");
  const [labAccounts, setLabAccounts] = useState(false);
  const [result, setResult] = useState(null);
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  const chosen = getSelectedChecks();
  const hasAuth = !!buildAuthPayload();

  async function handleValidate() {
    setError("");
    setResult(null);
    try {
      setResult(await validateTarget(url, permission));
    } catch (err) {
      setError(err.message);
    }
  }

  async function handleStart(e) {
    e.preventDefault();
    setError("");
    setBusy(true);
    try {
      const endpoints = parseEndpoints(endpointsText);
      const payload = {
        url,
        permission_confirmed: permission,
        lab_accounts: labAccounts,
      };
      if (endpoints.length) payload.endpoints = endpoints;
      const checkIds = getSelectedChecks();
      if (checkIds && checkIds.length) payload.check_ids = checkIds;
      const auth = buildAuthPayload();
      if (auth && !labAccounts) payload.auth = auth;
      const scan = await startScan(payload);
      rememberScan(scan.id);
      navigate("/progress/" + scan.id);
    } catch (err) {
      setError(err.message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div style={{ maxWidth: 640 }}>
      <h2>Target Setup</h2>

      <form onSubmit={handleStart}>
        <label>
          API address
          <br />
          <input
            type="text"
            placeholder="http://127.0.0.1:5000"
            value={url}
            onChange={(e) => setUrl(e.target.value)}
            style={{ width: "100%" }}
          />
        </label>

        <p>
          <label>
            Endpoints to test (one per line, optional)
            <br />
            <textarea
              rows={7}
              style={{ width: "100%", fontFamily: "monospace" }}
              placeholder={"/api/users/me\nGET /api/search q\n/api/orders/{id}\nPATCH /api/orders/{id}\nPOST /api/login"}
              value={endpointsText}
              onChange={(e) => setEndpointsText(e.target.value)}
            />
          </label>
          <small>
            Order matters: the first few endpoints are used by the config, input
            and rate checks. Use {"{id}"} for object paths.
          </small>
        </p>

        <p>
          <label>
            <input
              type="checkbox"
              checked={labAccounts}
              onChange={(e) => setLabAccounts(e.target.checked)}
            />{" "}
            Use the built-in fake lab accounts (lab on port 5000 only)
          </label>
        </p>

        <p>
          <label>
            <input
              type="checkbox"
              checked={permission}
              onChange={(e) => setPermission(e.target.checked)}
            />{" "}
            I own this system or have written permission to test it
          </label>
        </p>

        <p>
          <small>
            Checks to run:{" "}
            {chosen && chosen.length
              ? chosen.length + " selected on the Profile screen"
              : "all (change this on the Profile screen)"}
            <br />
            Logins:{" "}
            {labAccounts
              ? "built-in lab accounts"
              : hasAuth
              ? "set on the Auth screen (kept in memory only)"
              : "none (set them on the Auth screen)"}
          </small>
        </p>

        <button type="button" onClick={handleValidate} disabled={!url || !permission}>
          Validate target
        </button>{" "}
        <button type="submit" disabled={!url || !permission || busy}>
          {busy ? "Starting..." : "Start scan"}
        </button>
      </form>

      {error && <p style={{ color: "red" }}>{error}</p>}
      {result && <pre>{JSON.stringify(result, null, 2)}</pre>}
    </div>
  );
}

# API-Gapwolf

API-Gapwolf is a security scanner for web APIs. You give it the address of an API you own
(and optionally an OpenAPI file). It runs a limited set of non-destructive checks, shows
the findings in a dashboard, keeps the history in a local SQLite database, and exports a
redacted PDF report.

## Authorized use only

Only scan systems you own or have written permission to test. The report lists weaknesses,
so share it carefully. Results are evidence for review, not proof that an API is secure.

## What it checks

- Authentication: no login, made-up token, expired token, broken token, inconsistent login rules
- Authorization: object-level access (BOLA / IDOR), role check, PUT / PATCH / DELETE on another user's data
- Input and injection: SQL-injection indicators, wrong data types, boundary values, odd parameters, error replies
- Rate limiting: request limit, 429 answer, recovery
- Security configuration: HTTPS, security headers, CORS, sensitive data, error details

Logins: Bearer / JWT token, Basic, API key and OAuth 2.0 client credentials.

## Safety design

- The backend listens on 127.0.0.1 only.
- CORS allows only the exact dashboard address.
- Limits are enforced in code: 100 requests per scan, 2 parallel, 10 s timeout, 1 MB response.
- Uploads: 2 MB maximum, JSON or YAML only, safe YAML loading, remote references off.
- URLs: http or https only, no password inside the URL; private, loopback and metadata
  addresses are blocked. The one exception is the built-in lab (see below).
- Logins entered in the dashboard are kept in memory only and are cleared on page refresh.
- Reports and the database store masked evidence only. Report text is escaped.

## Requirements

- Python 3.14
- Node.js 22 or newer
- Windows PowerShell (the commands below use it)

## Quick start (one command)

Windows (PowerShell):

```
powershell -ExecutionPolicy Bypass -File .\start.ps1
```

Linux or macOS:

```
bash start.sh
```

The first run installs everything. Then it prints http://localhost:5173 and opens it in your
browser. Press Enter in that window to stop. Only this computer can open the dashboard.

Add `-Lab` (Windows) or `--lab` (Linux, macOS) to also start the practice lab. This turns the
lab exception on for that run only and does not change `backend\.env`. Without it, the lab
exception is always off. Add `-NoBrowser` or `--no-browser` to skip opening the browser.
Logs are written to the `logs` folder.

Needs Python 3, Node.js 22 or newer, and (Linux, macOS) curl.

## Run the demo with the practice lab

The `lab` folder contains an intentionally vulnerable API with fake data. It is never to be
exposed to a network. The lab answer key is in `docs\lab-answer-key.md`.

1. In `backend\.env` set `ALLOW_LAB_EXCEPTION=true`. This allows plain http for the local lab only.
   Set it back to `false` when you are not scanning the lab.
2. Start the lab, the backend and the dashboard in three terminals:

```
cd lab
..\.venv\Scripts\python.exe -m uvicorn lab_api:app --host 127.0.0.1 --port 5000
```
```
cd backend
..\.venv\Scripts\python.exe -m uvicorn app.main:app --host 127.0.0.1 --port 8000
```
```
cd frontend
npm run dev
```

3. Reset the lab before each scan:

```
Invoke-RestMethod -Method Post http://127.0.0.1:5000/lab/reset
```

4. Open http://localhost:5173. On the Auth screen set User A to Bearer with `lab-token-user-a`
   (expired token `lab-token-expired`) and User B to Bearer with `lab-token-user-b`.
5. On Target Setup enter `http://127.0.0.1:5000`, paste the endpoints, tick the permission box
   and start the scan.

The scan should find 9 of the 10 planted weaknesses. The tenth (no HTTPS) is flagged for manual
review because the lab exception allows plain http.

## Known limits

- Tested against the local lab only, not against real APIs.
- Findings marked suspected or manual review need a person to confirm them.
- The PUT check is skipped when the API has no PUT endpoint with an ID.
- Rate-limit checks use a safe size of 20 requests. Higher limits may exist.
- A missing security header is not always a weakness for a pure JSON API.
- The report shows "Passed" for PATCH and DELETE rows even when a weakness is found; the
  details column says "weakness found, merged into ...".
- Test results are in `docs\step24-test-record.md` and `docs\step25-self-check.md`.
